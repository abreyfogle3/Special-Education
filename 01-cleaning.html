shutdown_enrollment_numbers <- enrollment_numbers |>

    school_active = case_when(

Creating new columns to handle <10 Schools

enrollment_number_10 <- shutdown_enrollment_numbers |>
  mutate(
    x2025_2026 = case_when(
      x25_26 < 10 ~ NA,
      is.na(x25_26) ~ 0,
      .default = as.numeric(x25_26)),
     x2026_2027 = case_when(
      x26_27 < 10 ~ NA,
      is.na(x26_27) ~ 0,
      .default = as.numeric(x26_27),
    )
  )

Remove NA Rows and Remove old columns

removal_enrollment <- enrollment_number_10 |>
  filter(
    !is.na(x2025_2026),
    !is.na(x2026_2027)
  ) |>
  select(
    campus,
    school_active,
    x2025_2026,
    x2026_2027
  )
